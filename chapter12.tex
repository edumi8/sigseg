\section{Objectives and Scope}
\begin{itemize}
  \item Verify that the \ac{ISMS} and the corrective actions listed in
        Chapter~\ref{chap:activity12} conform to ISO/IEC 27001 requirements,
        the \ac{SoA}, and GreenGrid policies.
  \item Assess the \textbf{effectiveness} of controls implemented for
        \emph{shared assets} and for \emph{department-specific assets}.
  \item Identify new non-conformities (NC), observations (O), or
        opportunities for improvement (OFI) and assign corrective-action
        plans (CAPA) where required.
\end{itemize}

%----------------------------------------------------------------------
\section{Audit Methodology}
\begin{enumerate}
  \item \textbf{Document review}: policies, procedures, change records,
        CA logs.
  \item \textbf{Interviews}: asset owners, process owners, random sample
        of end-users.
  \item \textbf{Sampling \& testing}: log excerpts, configuration
        baselines, backup-restore evidence.
  \item \textbf{Evaluation}: compare evidence with ISO control objectives,
        \ac{SoA} inclusions, and risk-treatment decisions.
  \item \textbf{Reporting}: classify findings, raise or update CA tickets,
        obtain management acknowledgement.
\end{enumerate}

\section{Audit Conclusion}
\begin{enumerate}
  \item Controls were \emph{generally effective}; four minor and two major
        non-conformities were detected.
  \item No evidence of systemic failure, but privileged-access governance
        and cloud-logging require urgent attention.
  \item Follow-up audit focusing on CA-003, CA-005, and CA-009 is scheduled
        for \textbf{15 October 2025}.
\end{enumerate}
